# With 4 MiB files and bs=2m, each file holds exactly 2 blocks. iodepth=16

# the same files also failed with 22. The failure boundary at exactly
# iodepth/blocks_per_file = 16/2 = 8 files, identical in all 64 jobs, shows
# this is a fio DFS-ioengine limitation with I/O depth spanning file
# rotation - not a DAOS pool/capacity/network issue.
#
